\begin{center}{\LARGE\bfseries\color{NavyBlue} Answer Key}\end{center}
\vspace{1mm}

\begin{tcolorbox}[colback=Soft,colframe=Accent,title={\bfseries Drill A answers},fonttitle=\bfseries\color{white},coltitle=white,colbacktitle=Accent]
\textbf{1. Hook:} ``Everyone shops the market \emph{minus themselves}; add up what they grab; never beat the smaller pile.''\\[3pt]
\textbf{2. Formula:} $\displaystyle M=\min\!\Big(\sum_c\min(b_c,\ S-s_c),\ \min(B,S)\Big)$\\[3pt]
\textbf{3. Loop:} \code{reach=0; for c: reach += min(c.buy, totalSell - c.sell); return min(reach, min(B,S));}\\[3pt]
\textbf{4. $-s_c$:} a firm may not use its \emph{own} sells, so its allowed pool is all sells minus its own --- that enforces the different-company rule.\\[3pt]
\textbf{5. Answer $=130$.} $B{=}240$, $S{=}240$. A: $\min(200,240{-}150){=}90$; B: $\min(40,240){=}40$; C: $0$. Sum $=130 \le \min(B,S){=}240$, so the \emph{reach} binds. It's not $240$ because A's big buy may only reach the $90$ of sells that aren't A's own.
\end{tcolorbox}

\begin{tcolorbox}[colback=Soft,colframe=NavyBlue,title={\bfseries Drill C answers},fonttitle=\bfseries\color{white},coltitle=white,colbacktitle=NavyBlue]
\footnotesize
\textbf{Structures:} \code{orders\_} (id$\to$Order, source of truth; makes cancelById/getAll fast) $\cdot$ \code{ordersByUser\_} (user$\to$ids; cancelForUser) $\cdot$ \code{ordersBySecurity\_} (sec$\to$ids; the per-sec cancel) $\cdot$ \code{securityTotals\_} (sec$\to$\{B,S,per-company b/s\}; matching).\\[3pt]
\textbf{Complexity:} add/cancel $O(1)$; bulk cancels $O(k)$; matching $O(\#\text{companies})$; getAll $O(n)$.\\[3pt]
\textbf{Why two helpers} (\code{insertOrderInternal}/\code{removeOrderInternal}): every write funnels through them, so the four structures update in \emph{one place} and can't desync.
\end{tcolorbox}
