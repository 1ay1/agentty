# Header-side allowlist for the concurrency ban-list (see allowlist.txt for
# the rationale; this file covers include/, that one covers src/).
#
# Paths are relative to include/.

# ── Types whose whole job is to be the safe wrapper ──────────────────
# NOTE: util/background.hpp (was isolated_thread.hpp) is deliberately NOT
# here any more. It used to need `thread detach` because it hand-rolled a
# detached worker, exempted on the grounds that the ACP process mode has no
# jaal loop and therefore no task machinery. That premise was wrong:
# maya::pool is standalone and needs no loop, so background.hpp now
# posts to it and spawns no thread of its own.
# ranked_lock: lock-ordering enforcement. The mutex IS the subject, and the
# thread_local is the per-thread rank stack it checks against.
agentty/util/ranked_lock.hpp: mutex thread_local
agentty/util/logx.hpp: atomic

# ── The turn's cancel token ──────────────────────────────────────────────
# agentty/io/http.hpp has NO entry. http::CancelToken is still the app's own
# (a reducer tripping a specific in-flight request from a later step — see the
# Sendable opt-in in runtime/cmd.hpp), but it is a std::stop_source now, not
# an atomic flag, so it can hand out a real stop_token and needs no exemption.
# backoff_with_jitter(): per-thread mt19937 for retry jitter. The generator
# holds no shared state; each caller thread gets its own seeded from
# random_device. thread_local is what makes this branch-free at the call
# site without a per-turn allocation.
agentty/provider/error_class.hpp: thread_local

# ── Caches and registries below the loop ─────────────────────────────────
agentty/domain/catalog.hpp: mutex atomic
agentty/domain/lazy_bytes.hpp: atomic
agentty/domain/ui_live.hpp: mutex
agentty/domain/ui_theme.hpp: atomic
# const_cast on a std::variant alternative accessor, not on shared state.
agentty/domain/smart_mode.hpp: const_cast

# ── ACP: a separate process mode, no jaal loop ───────────────────────────
agentty/acp/server.hpp: atomic
